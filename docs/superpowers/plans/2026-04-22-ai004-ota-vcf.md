# AI004 OTA VCF VCV Rack Plugin Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a fully functional VCV Rack v2 plugin implementing the AI004 OTA VCF (MS-20-style filter) as a single 8HP module with LP/HP modes, CV input, and two panel skins.

**Architecture:** A self-contained `OtaVcf` DSP class (zero SDK dependencies) implements a two-stage OTA filter with nonlinear feedback, owned by the `AI004Module` Rack module struct that wires knob/CV inputs to DSP params each audio block. The `AI004Widget` class handles SVG panel switching via a right-click context menu, persisting the selection through `dataToJson`/`dataFromJson`.

**Tech Stack:** C++17, VCV Rack v2 SDK, VCV Rack plugin Makefile (wraps CMake), no external DSP libraries, standalone C++ unit tests (no framework dependency).

    float sum = 0.f;
    for (int i = 0; i < n; ++i) sum += buf[i] * buf[i];
    return std::sqrt(sum / n);
}

static float sine_sample(float freq_hz, int i) {
    return std::sin(2.f * 3.14159265f * freq_hz * static_cast<float>(i) / kSampleRate);
}

static void test_lp_passes_low_freq() {

    for (int i = 0; i < kNumSamples; ++i)
        out[i] = vcf.process(sine_sample(100.f, i), 1000.f, 0.f, kSampleRate, OtaVcf::Mode::LP);
    float amplitude = rms(out + kNumSamples / 2, kNumSamples / 2);  // skip transient
    assert(amplitude > 0.3f);
    printf("PASS: LP passes low frequency (rms=%.3f)\n", amplitude);
}

- [ ] **Step 1: Create `src/OtaVcf.hpp`**

```cpp
#pragma once
#include <cmath>
#include <algorithm>

// Two-stage OTA ladder filter modeling the AI004 / Korg MS-20 circuit.
//
// Topology: input → soft saturator → one-pole LP (stage 1) → one-pole LP (stage 2)
//           with feedback from stage 2 output through an asymmetric diode pair.
//
// LP output: stage 2 (12dB/oct).
// HP output: input minus stage 1 output (6dB/oct) — matches hardware spec.
//
// No SDK dependencies — safe to include in standalone unit tests.
class OtaVcf {
public:
    enum class Mode { LP, HP };

    OtaVcf() { reset(); }

    void reset() { lp1_ = 0.f; lp2_ = 0.f; }

    // Process one audio sample.
    //   in          — input sample (any level; ±5V typical in VCV Rack)
    //   cutoff_hz   — cutoff frequency in Hz, clamped to [20, 20000]
    //   resonance   — 0..1; self-oscillation begins around 0.8
    //   sample_rate — host sample rate in Hz
    //   mode        — LP (12dB/oct output) or HP (6dB/oct output)
    float process(float in, float cutoff_hz, float resonance,
                  float sample_rate, Mode mode) {
        float g = computeG(cutoff_hz, sample_rate);
        float k = resonance * kMaxK;

        // Feedback through asymmetric diode pair — this is the source of MS-20 character.
        // The asymmetry (different gain for + and - half-cycles) creates even harmonics.
        float fb = diodeSaturation(k * lp2_);

        // Soft saturation at the OTA input stage — limits overdrive, prevents instability.
        float driven = softSaturate(in - fb);

        // Cascade two one-pole LP stages.
        float stage1 = processOnePole(lp1_, driven, g);
        float stage2 = processOnePole(lp2_, stage1, g);

        // HP is taken before stage 2: one-pole HP = input minus one-pole LP.
        return (mode == Mode::LP) ? stage2 : (in - stage1);
    }

private:
    // kMaxK = 4.0: at resonance=1.0, loop gain just exceeds 1 near cutoff → self-oscillation.
    // At resonance=0.8, loop gain ≈ 0.8 * 4.0 * (0.5)^2 = 0.8, approaching but not exceeding 1.
    // Empirically this places self-oscillation onset around 80–90% resonance.
    static constexpr float kMaxK     = 4.0f;
    static constexpr float kDiodePos = 1.4f;  // forward gain: more drive on positive half
    static constexpr float kDiodeNeg = 0.8f;  // reverse gain: softer on negative half

    float lp1_ = 0.f;  // one-pole stage 1 state
    float lp2_ = 0.f;  // one-pole stage 2 state

    // Bilinear-warped frequency coefficient.
    // Clamping fc before tan() prevents instability near Nyquist.
    static float computeG(float cutoff_hz, float sample_rate) {
        float fc = std::clamp(cutoff_hz, 20.f, 20000.f) / sample_rate;
        fc = std::clamp(fc, 0.001f, 0.499f);
        return std::tan(3.14159265f * fc);
    }

    // One-pole LP forward-Euler step. Updates state in-place, returns new output.
    // The 1/(1+g) normalization keeps the stage from saturating with high g values.
    static float processOnePole(float& state, float x, float g) {
        float gn = g / (1.f + g);
        state += gn * (x - state);
        return state;
    }

    // tanh-based input saturation models the OTA transconductance curve.
    static float softSaturate(float x) { return std::tanh(x); }

    // Asymmetric diode pair: forward-biased diode clips harder on positive half,
    // giving the characteristic MS-20 "gnarly" resonance sound.
    static float diodeSaturation(float x) {
        if (x >= 0.f) return std::tanh(x * kDiodePos);
        else          return std::tanh(x * kDiodeNeg);
    }
};
```

```cpp
#include "plugin.hpp"
#include "OtaVcf.hpp"

// ─── Module struct ─────────────────────────────────────────────────────────────

struct AI004Module : rack::Module {
    enum ParamId {
        CUTOFF_PARAM,
        RESONANCE_PARAM,
        CVAMT_PARAM,
        LPSWITCH_PARAM,
        PARAMS_LEN
    };
    enum InputId {
        AUDIO_IN,
        CV_IN,
        INPUTS_LEN
    };
    enum OutputId {
        AUDIO_OUT,
        OUTPUTS_LEN
    };

    OtaVcf vcf;
    int panel_skin = 0;  // 0 = aluminum, 1 = black

    AI004Module() {
        config(PARAMS_LEN, INPUTS_LEN, OUTPUTS_LEN);

        // Cutoff stored as log2(Hz) so the knob maps linearly to octaves.
        // display base=2 causes Rack to render the tooltip in Hz automatically.
        configParam(CUTOFF_PARAM, std::log2(20.f), std::log2(20000.f),
                    std::log2(1000.f), "Cutoff", " Hz", 2.f);
        configParam(RESONANCE_PARAM, 0.f, 1.f, 0.f, "Resonance", "%", 0.f, 100.f);
        configParam(CVAMT_PARAM, -1.f, 1.f, 0.f, "CV Amount");
        configSwitch(LPSWITCH_PARAM, 0.f, 1.f, 0.f, "Filter mode", {"LP", "HP"});

        configInput(AUDIO_IN, "Audio");
        configInput(CV_IN, "CV");
        configOutput(AUDIO_OUT, "Audio");
    }

    void process(const rack::Module::ProcessArgs& args) override {
        float cutoff_hz = computeCutoffHz();
        float resonance = params[RESONANCE_PARAM].getValue();
        OtaVcf::Mode mode = (params[LPSWITCH_PARAM].getValue() >= 0.5f)
                          ? OtaVcf::Mode::HP : OtaVcf::Mode::LP;

        float in  = inputs[AUDIO_IN].getVoltage();
        float out = vcf.process(in, cutoff_hz, resonance, args.sampleRate, mode);
        outputs[AUDIO_OUT].setVoltage(out);
    }

    json_t* dataToJson() override {
        json_t* root = json_object();
        json_object_set_new(root, "panel_skin", json_integer(panel_skin));
        return root;
    }

    void dataFromJson(json_t* root) override {
        json_t* skin = json_object_get(root, "panel_skin");
        if (skin) panel_skin = static_cast<int>(json_integer_value(skin));
    }

private:
    // CV is 1V/oct: adding CV voltage (scaled by CV Amount) to log2(fc) shifts by octaves.
    float computeCutoffHz() const {
        float log2_fc = params[CUTOFF_PARAM].getValue();
        float cv_v    = inputs[CV_IN].getVoltage();
        float cv_amt  = params[CVAMT_PARAM].getValue();
        float fc      = std::pow(2.f, log2_fc + cv_amt * cv_v);
        return rack::clamp(fc, 20.f, 20000.f);
    }
};
```

            [m] { return m->panel_skin == 0; },
            [m] { m->panel_skin = 0; }));
        menu->addChild(rack::createCheckMenuItem("Black", "",
            [m] { return m->panel_skin == 1; },
            [m] { m->panel_skin = 1; }));
    }
